### Host builds for the differential test (NOT used for size numbers) -- see build_host.sh
# hostinc/regex.h = stub with musl's regex_t layout so regcomp.c/regexec.c/tre-mem.c compile against the macOS libc
# (musl's own include tree cannot be used on the host: mbtowc/iswctype ABI/types differ). driver.c = CLI contract.
sh build_host.sh
#   driver        : plain host libc, LC_CTYPE=en_US.UTF-8 (Unicode-aware \w \s and case folding, as musl's C.UTF-8 would)
#   driver_ascii  : engine built with -D_DONT_USE_CTYPE_INLINE_ -Diswalnum=ascii_iswalnum ... -> asciishim.c (ASCII-only classes/case)
#   memtrace      : engine built with -Dmalloc=tr_malloc ... -> counts heap (host, 64-bit)
python3 $S/tools/difftest.py ./driver       $S/tools/corpus_seed1.jsonl 5000 > difftest_full.txt     # cases=2088 checked=2081 pass=1406 fail=675 timeouts=0
MUSL_REG_NEWLINE=1 python3 $S/tools/difftest.py ./driver $S/tools/corpus_seed1.jsonl 5000 > difftest_newline.txt   # same totals: pass=1406 (2 '.'-newline cases fixed, 2 anchor cases broken)
python3 $S/tools/difftest.py ./driver_ascii $S/tools/corpus_seed1.jsonl 5000 > difftest_ascii.txt    # cases=2088 checked=2081 pass=1436 fail=645 timeouts=0
python3 classify.py ./driver ./driver_ascii > classify_out.txt   # failure classification (both builds) + per-bucket examples
sh -c 'while IFS="$(printf "\t")" read -r p t; do ./driver "$p" "$t"; done < probes.tsv' > probes_out.txt   # hand probes of syntax/semantics
# heap: see heap_out.txt (./memtrace PATTERN TEXT)
# ARM struct sizes: sz.c compiled with the ARM include set -> nm -S: tre_tnfa_transition_t=32 B, tre_tnfa_t=68 B, regex_t=32 B, tre_mem_struct=24 B, tre_submatch_data_t=12 B
sh deps_size.sh > deps_size.txt   # Cortex-M4 size of musl's own implementations of the nm -u symbols
